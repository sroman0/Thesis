\section{Process Creation}
\label{sec:appendix-process-creation}

\begin{lstlisting}[
    language=C,
    basicstyle=\ttfamily\scriptsize,
    caption={Entry and return programs used to construct the
    \texttt{clone} event. Repeated serialisation calls are omitted.},
    label={lst:appendix-clone-pair}
]
SEC("tracepoint/syscalls/sys_enter_clone")
int tracepoint__syscalls__sys_enter_clone(sys_enter_clone_ctx_t *ctx)
{
    args_t args = {};
    args.args[0] = ctx->flags;
    args.args[1] = ctx->stack;
    args.args[2] = ctx->parent_tid;
    args.args[3] = ctx->child_tid;
    args.args[4] = ctx->tls;
    return save_args(&args, CLONE);
}

SEC("tracepoint/syscalls/sys_exit_clone")
int tracepoint__syscalls__sys_exit_clone(sys_exit_generic_ctx_t *ctx)
{
    args_t args = {};
    if (load_args(&args, CLONE) != 0)
        return 0;
    del_args(CLONE);

    program_data_t p = {};
    if (!init_program_data(&p, ctx, CLONE))
        return 0;

    unsigned long flags = args.args[0];
    long ret = ctx->ret;
    save_to_submit_buf(&p.event->args_buf,
                       &flags, sizeof(unsigned long), 0);
    /* stack, TID pointers and TLS occupy indexes 1--4. */
    save_to_submit_buf(&p.event->args_buf,
                       &ret, sizeof(long), 5);
    return events_perf_submit(&p);
}
\end{lstlisting}

\section{Executable Transitions}
\label{sec:appendix-executable-transitions}

\begin{lstlisting}[
    language=C,
    basicstyle=\ttfamily\scriptsize,
    caption={Collection of \texttt{execve} and \texttt{execveat}
    requests at syscall entry.},
    label={lst:appendix-exec-requests}
]
SEC("tracepoint/syscalls/sys_enter_execve")
int tracepoint__syscalls__sys_enter_execve(sys_enter_execve_ctx_t *ctx)
{
    program_data_t p = {};
    if (!init_program_data(&p, ctx, EXECVE) || !ctx->pathname)
        return 0;

    save_str_to_buf(&p.event->args_buf, (void *) ctx->pathname, 0);
    if (ctx->argv)
        save_str_arr_to_buf(
            &p.event->args_buf,
            (const char __user *const __user *) ctx->argv, 1);
    return events_perf_submit(&p);
}

SEC("tracepoint/syscalls/sys_enter_execveat")
int tracepoint__syscalls__sys_enter_execveat(sys_enter_execveat_ctx_t *ctx)
{
    program_data_t p = {};
    if (!init_program_data(&p, ctx, EXECVEAT))
        return 0;

    int dirfd = ctx->dirfd;
    int flags = ctx->flags;
    save_to_submit_buf(&p.event->args_buf,
                       &dirfd, sizeof(int), 0);
    if (ctx->pathname)
        save_str_to_buf(&p.event->args_buf, (void *) ctx->pathname, 1);
    save_to_submit_buf(&p.event->args_buf,
                       &flags, sizeof(int), 2);
    if (ctx->argv)
        save_str_arr_to_buf(
            &p.event->args_buf,
            (const char __user *const __user *) ctx->argv, 3);
    return events_perf_submit(&p);
}
\end{lstlisting}

\begin{lstlisting}[
    language=C,
    basicstyle=\ttfamily\scriptsize,
    caption={Resolved binary identity and successful executable transition,
    abridged from \texttt{security\_bprm\_check} and
    \texttt{sched\_process\_exec}.},
    label={lst:appendix-exec-kernel-stages}
]
/* security_bprm_check: resolved executable under mediation. */
if (!bprm)
    return 0;
struct file *file = BPF_CORE_READ(bprm, file);
if (!file)
    return 0;

void *file_path = get_path_str(
    __builtin_preserve_access_index(&file->f_path));
if (!file_path)
    return 0;

dev_t s_dev = get_dev_from_file(file);
unsigned long inode_nr = get_inode_nr_from_file(file);
save_str_to_buf(&p.event->args_buf, file_path, 0);
save_to_submit_buf(&p.event->args_buf,
                   &s_dev, sizeof(dev_t), 1);
save_to_submit_buf(&p.event->args_buf,
                   &inode_nr, sizeof(unsigned long), 2);
/* filename, argc and envc occupy indexes 3--5. */

/* sched_process_exec: replacement has completed. */
struct linux_binprm *completed_bprm =
    (struct linux_binprm *) ctx->args[2];
if (!completed_bprm)
    return 0;
const char *filename = BPF_CORE_READ(completed_bprm, filename);
if (!filename)
    return 0;

save_str_to_buf(&p.event->args_buf, (void *) filename, 0);
events_perf_submit(&p);
\end{lstlisting}

\section{Credential Transitions}
\label{sec:appendix-credential-transition}

\begin{lstlisting}[
    language=C,
    basicstyle=\ttfamily\scriptsize,
    caption={Change-only collection in
    \texttt{security\_task\_fix\_setuid}.},
    label={lst:appendix-fix-setuid}
]
u32 old_uid = BPF_CORE_READ(old_cred, uid.val);
u32 new_uid = BPF_CORE_READ(new_cred, uid.val);
u32 old_euid = BPF_CORE_READ(old_cred, euid.val);
u32 new_euid = BPF_CORE_READ(new_cred, euid.val);
u32 old_suid = BPF_CORE_READ(old_cred, suid.val);
u32 new_suid = BPF_CORE_READ(new_cred, suid.val);
u32 old_fsuid = BPF_CORE_READ(old_cred, fsuid.val);
u32 new_fsuid = BPF_CORE_READ(new_cred, fsuid.val);

if (old_uid == new_uid &&
    old_euid == new_euid &&
    old_suid == new_suid &&
    old_fsuid == new_fsuid)
    return 0;

save_to_submit_buf(&p.event->args_buf,
                   &old_uid, sizeof(u32), 0);
save_to_submit_buf(&p.event->args_buf,
                   &new_uid, sizeof(u32), 1);
save_to_submit_buf(&p.event->args_buf,
                   &old_euid, sizeof(u32), 2);
save_to_submit_buf(&p.event->args_buf,
                   &new_euid, sizeof(u32), 3);
/* saved and filesystem UIDs occupy indexes 4--7. */
save_to_submit_buf(&p.event->args_buf,
                   &flags, sizeof(int), 8);
return events_perf_submit(&p);
\end{lstlisting}
